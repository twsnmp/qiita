# go-sixelパッケージ


https://github.com/mattn/go-sixel/tree/master


# サンプルプログラム

```go:main.go
package main

import (
	"image"
	_ "image/gif"
	_ "image/jpeg"
	_ "image/png"
	"log"
	"os"

	"github.com/mattn/go-sixel"
)

func main() {
	if len(os.Args) != 2 {
		log.Fatalln("Usage: sixel [image file]")
	}
	fp, err := os.Open(os.Args[1])
	if err != nil {
		log.Fatalln(err)
	}
	img, _, err := image.Decode(fp)
	if err != nil {
		log.Fatalln(err)
	}
	sixel.NewEncoder(os.Stdout).Encode(img)
}

```

指定したファイルをsixelに変換してターミナルに表示できます。
