# はじめに

OpenTelemetryのプロトコルでテレメトリーデータ（メトリクス、ログ、トレース）を送受信するテストプログラムを作りました。その紹介です。


![image.png](https://qiita-image-store.s3.ap-northeast-1.amazonaws.com/0/3874862/bc2b35b8-2da4-4a71-9735-b77e9a3f40f4.png)



# OpenTelemetry

OpenTelemetryは、ソフトウェアのパフォーマンスや動作の分析のための技術です。
テレメトリーデータ（メトリクス、ログ、トレース）の計装、生成、収集、エクスポートを実現するための
API、SDK、ツールを提供しています。

クラウドサービスなどを観測可能する技術です。

https://opentelemetry.io/ja/


# 受信側

OpenTelemetryのコレクター

https://opentelemetry.io/ja/docs/collector/

で受信する例は多数ありますが、受信するプログラムの例はほとんどありません。
コレクターのソースコード

https://github.com/open-telemetry/opentelemetry-collector


にOTLPの受信パッケージがあります。

https://github.com/open-telemetry/opentelemetry-collector/tree/main/receiver/otlpreceiver

このパッケージを使って受信プログラムを作りました。

```go:main.go
package main

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"log"

	"go.opentelemetry.io/collector/client"
	"go.opentelemetry.io/collector/component"
	"go.opentelemetry.io/collector/component/componenttest"
	"go.opentelemetry.io/collector/consumer"
	"go.opentelemetry.io/collector/pdata/plog"
	"go.opentelemetry.io/collector/pdata/pmetric"
	"go.opentelemetry.io/collector/pdata/ptrace"
	"go.opentelemetry.io/collector/receiver"
	"go.opentelemetry.io/collector/receiver/otlpreceiver"
)

func main() {
	f := otlpreceiver.NewFactory()
	config := f.CreateDefaultConfig()
	if c, ok := config.(*otlpreceiver.Config); ok {
		c.HTTP.ServerConfig.Endpoint = "0.0.0.0:4318"
	}
	log.Printf("config=%#v", config)
	componentID := component.MustNewID("otlp")
	ctx := context.Background()
	// METRICS
	log.Println("start metrics")
	mr, err := consumer.NewMetrics(func(ctx context.Context, md pmetric.Metrics) error {
		f := client.FromContext(ctx)
		log.Printf("recv metric from %s", f.Addr.String())
		jm := pmetric.JSONMarshaler{}
		if js, err := jm.MarshalMetrics(md); err == nil {
			printJSON(js)
		} else {
			log.Printf("err=%v", err)
		}
		return nil
	})
	if err != nil {
		log.Fatalln(err)
	}
	_, err = f.CreateMetrics(ctx,
		receiver.Settings{
			ID:                componentID,
			TelemetrySettings: componenttest.NewNopTelemetrySettings(),
		},
		config, mr)
	if err != nil {
		log.Fatalf("Failed to create receiver otlp: %v", err)
	}
	// TRACES
	log.Println("start traces")
	tr, err := consumer.NewTraces(func(ctx context.Context, td ptrace.Traces) error {
		f := client.FromContext(ctx)
		log.Printf("recv trace from %s", f.Addr.String())
		jm := ptrace.JSONMarshaler{}
		if js, err := jm.MarshalTraces(td); err == nil {
			printJSON(js)
		} else {
			log.Printf("err=%v", err)
		}
		return nil
	})
	if err != nil {
		log.Fatalln(err)
	}
	_, err = f.CreateTraces(ctx,
		receiver.Settings{
			ID:                componentID,
			TelemetrySettings: componenttest.NewNopTelemetrySettings(),
		},
		config, tr)
	if err != nil {
		log.Fatalf("Failed to create receiver otlp: %v", err)
	}
	// LOGS
	log.Println("start logs")
	lr, err := consumer.NewLogs(func(ctx context.Context, ld plog.Logs) error {
		f := client.FromContext(ctx)
		log.Printf("recv log form %s", f.Addr.String())
		jm := plog.JSONMarshaler{}
		if js, err := jm.MarshalLogs(ld); err == nil {
			printJSON(js)
		} else {
			log.Printf("err=%v", err)
		}
		return nil
	})
	if err != nil {
		log.Fatalln(err)
	}
	logReceiver, err := f.CreateLogs(ctx,
		receiver.Settings{
			ID:                componentID,
			TelemetrySettings: componenttest.NewNopTelemetrySettings(),
		},
		config, lr)
	if err != nil {
		log.Fatalf("Failed to create receiver otlp: %v", err)
	}
	err = logReceiver.Start(ctx, nil)
	if err != nil {
		log.Fatalln(err)
	}
	// Keep the server running
	select {}
}

func printJSON(j []byte) {
	var dst bytes.Buffer
	if err := json.Indent(&dst, j, "", "  "); err == nil {
		fmt.Println(dst.String())
		return
	}
	fmt.Println(string(j))
}

```

ポイントは、

- `otlpreceiver.NewFactor`でFactoryを作る
- Factoryでデフォルト設定を作る
- デフォルト設定を変更する
- Metric,Trace,Logの処理ルーチンをconsumerパッケージで作る
- Factoryで各受信サーバーを作る
- 受信サーバーを変更した設定で起動する

起動は、最後のログサーバーだけ
```go
	err = logReceiver.Start(ctx, nil)
	if err != nil {
		log.Fatalln(err)
	}
```
にします。MetriやTraceを起動してしまうと、後から登録した処理が
動作しません。


# 送信側

送信側は、各言語のSDKにGetting Startedがあります。Go言語の場合は

https://opentelemetry.io/ja/docs/languages/go/getting-started/

です。

このままでは、画面にメトリックなどが表示されるので、受信側にOTLPで送信するように、
このソースコードを少し変更します。

```diff_go:otel.go

func newtracerProvider() (*trace.TracerProvider, error) {
-	 traceExporter, err := stdouttrace.New(
-		stdouttrace.WithPrettyPrint())
+	traceExporter, err := otlptracehttp.New(context.Background(), otlptracehttp.WithEndpointURL("http://localhost:4318/v1/traces"))

	if err != nil {
		return nil, err
	}

	tracerProvider := trace.NewTracerProvider(
		trace.WithBatcher(traceExporter,
			// Default is 5s. Set to 1s for demonstrative purposes.
			trace.WithBatchTimeout(5*time.Second)),
	)
	return tracerProvider, nil
}

func newMeterProvider() (*metric.MeterProvider, error) {
- metricExporter, err := stdoutmetric.New()
+	metricExporter, err := otlpmetrichttp.New(context.Background(), otlpmetrichttp.WithEndpointURL("http://localhost:4318/v1/metrics"), otlpmetrichttp.WithInsecure())
	if err != nil {
		return nil, err
	}

	meterProvider := metric.NewMeterProvider(
		metric.WithReader(metric.NewPeriodicReader(metricExporter,
			// Default is 1m. Set to 3s for demonstrative purposes.
			metric.WithInterval(5*time.Second))),
	)
	return meterProvider, nil
}

func newLoggerProvider() (*log.LoggerProvider, error) {
-	logExporter, err := stdoutlog.New()
+	logExporter, err := otlploghttp.New(context.Background(), otlploghttp.WithEndpointURL("http://localhost:4318/v1/logs"))
	if err != nil {
		return nil, err
	}

	loggerProvider := log.NewLoggerProvider(
		log.WithProcessor(log.NewBatchProcessor(logExporter)),
	)
	return loggerProvider, nil
}


```

受信側と受信側のプログラムを起動します。
他のターミナルから

```terminal
$curl http://localhost:8080/rolldice/
```

のようにアクセスします。
